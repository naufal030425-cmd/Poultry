    <script>
        // 1. Core Model Functions
        function evaluateWeightFunction(x) {
            return (-0.012 * Math.pow(x, 3)) + (1.45 * Math.pow(x, 2)) + (12 * x) + 45;
        }

        function evaluateDerivativeFunction(x) {
            return (-0.036 * Math.pow(x, 2)) + (2.90 * x) + 12;
        }

        // 2. Data Arrays Construction
        const timelineLabels = [];
        const cumulativeWeightData = [];
        const growthVelocityData = [];

        for (let day = 0; day <= 42; day++) {
            timelineLabels.push(day);
            cumulativeWeightData.push(evaluateWeightFunction(day).toFixed(1));
            growthVelocityData.push(evaluateDerivativeFunction(day).toFixed(1));
        }

        // 3. Chart.js Execution
        const ctx = document.getElementById('poultryTelemetryChart').getContext('2d');
        const poultryTelemetryChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: timelineLabels,
                datasets: [
                    {
                        label: 'Cumulative Biomass f(x) (grams)',
                        data: cumulativeWeightData,
                        borderColor: '#06b6d4', // Cyan
                        backgroundColor: 'rgba(6, 180, 212, 0.05)',
                        borderWidth: 3,
                        yAxisID: 'yAxisL',
                        tension: 0.2,
                        pointBackgroundColor: '#06b6d4',
                        pointHoverRadius: 6
                    },
                    {
                        label: "Growth Acceleration f'(x) (g/day)",
                        data: growthVelocityData,
                        borderColor: '#10b981', // Emerald
                        backgroundColor: 'transparent',
                        borderWidth: 2,
                        borderDash: [4, 4],
                        yAxisID: 'yAxisR',
                        tension: 0.2,
                        pointBackgroundColor: '#10b981',
                        pointHoverRadius: 6
                    }
                ]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                interaction: { mode: 'index', intersect: false },
                plugins: {
                    legend: {
                        position: 'top',
                        labels: { boxWidth: 12, font: { size: 11, weight: 'bold' }, color: '#334155' }
                    }
                },
                scales: {
                    x: {
                        title: { display: true, text: 'Broiler Production Lifespan (Days)', color: '#64748b', font: { size: 11, weight: 'bold' } },
                        grid: { color: '#f1f5f9' }
                    },
                    yAxisL: {
                        type: 'linear',
                        position: 'left',
                        title: { display: true, text: 'Cumulative Weight (g)', color: '#06b6d4', font: { size: 11, weight: 'bold' } },
                        grid: { color: '#f1f5f9' }
                    },
                    yAxisR: {
                        type: 'linear',
                        position: 'right',
                        title: { display: true, text: 'Instantaneous Change Velocity (g/day)', color: '#10b981', font: { size: 11, weight: 'bold' } },
                        grid: { drawOnChartArea: false }
                    }
                },
                onClick: (event, elements) => {
                    if (elements.length > 0) {
                        const targetIndex = elements[0].index;
                        processTelemetryNode(targetIndex);
                    }
                }
            }
        });

        // 4. Operational Diagnostics Controller
        function processTelemetryNode(dayIndex) {
            const day = timelineLabels[dayIndex];
            const weight = evaluateWeightFunction(day);
            const derivative = evaluateDerivativeFunction(day);

            // Display UI updates
            document.getElementById('promptState').classList.add('hidden');
            const panel = document.getElementById('telemetryPanel');
            const outputState = document.getElementById('analyticsState');
            outputState.classList.remove('hidden');

            // Apply calculated elements
            document.getElementById('lblDay').innerText = `Day ${day}`;
            document.getElementById('lblWeight').innerText = `Total Weight: ${weight.toFixed(1)} g`;
            document.getElementById('lblDerivative').innerText = `+${derivative.toFixed(2)}`;

            // UI Elements variables for style manipulation
            const badge = document.getElementById('badgeStatus');
            const explanation = document.getElementById('lblExplanation');

            // Categorization and interactive conditional routing logic
            if (day <= 10) {
                panel.className = "bg-white rounded-2xl border-2 border-cyan-500 p-6 shadow-sm transition-all duration-300";
                badge.className = "text-[10px] font-extrabold px-2 py-0.5 rounded-full uppercase bg-cyan-100 text-cyan-800";
                badge.innerText = "Brooding Phase";
                explanation.innerHTML = `During this early stage, the derivative value is low but rising. The chick is dedicating its metabolic energy to forming organs and skeletal structure. It is eating very little, but its rate of gain is accelerating day by day.`;
            } else if (day > 10 && day <= 28) {
                panel.className = "bg-white rounded-2xl border-2 border-emerald-500 p-6 shadow-sm transition-all duration-300";
                badge.className = "text-[10px] font-extrabold px-2 py-0.5 rounded-full uppercase bg-emerald-100 text-emerald-800";
                badge.innerText = "Peak Conversion Efficiency";
                explanation.innerHTML = `The derivative value is surging dramatically, indicating high growth velocity. The bird is packing on an impressive <strong>${derivative.toFixed(1)} grams of body mass every 24 hours</strong>. Your feed formulation should be rich and precisely optimized right now to capitalize on this high-efficiency muscle-building window.`;
            } else if (day > 28 && day <= 37) {
                panel.className = "bg-white rounded-2xl border-2 border-amber-500 p-6 shadow-sm transition-all duration-300";
                badge.className = "text-[10px] font-extrabold px-2 py-0.5 rounded-full uppercase bg-amber-100 text-amber-800";
                badge.innerText = "Diminishing Returns Inflection";
                explanation.innerHTML = `Look closely at the curves: though the total mass curve is still rising, the derivative curve has flattened and is now slanting down. This means the <em>rate</em> of growth is decelerating. The bird is beginning to use more feed for body maintenance and fat deposition rather than building muscle mass.`;
            } else {
                panel.className = "bg-white rounded-2xl border-2 border-rose-500 p-6 shadow-sm transition-all duration-300";
                badge.className = "text-[10px] font-extrabold px-2 py-0.5 rounded-full uppercase bg-rose-100 text-rose-800";
                badge.innerText = "Critical Harvest Window";
                explanation.innerHTML = `The derivative value has dropped sharply. The broiler has hit its biological plateau and is consuming massive amounts of feed simply to sustain its current size. Keeping the birds active past this window wastes expensive feed and heavily damages your overall Feed Conversion Ratio (FCR). It is time to harvest and ship to the market.`;
            }
        }
